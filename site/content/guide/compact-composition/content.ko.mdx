## 현재 페이지와 전체 결과 선택 \{#selection-scope\}

체크박스는 현재 페이지의 25건에 적용됩니다. 별도 버튼은 일치하는 800건 전체를 선택합니다. 페이지 목록을 펼쳐 한 항목을 해제하고 합계를 확인합니다.

<ComponentPreview name="selection-scope" title="현재 페이지와 전체 결과 선택" height={230} narrowHeight={280} background="plain" bodyClass="flex flex-col justify-center p-md" html={`<section aria-label="Recording selection">
  <label class="check"><input class="checkbox" type="checkbox" id="page-select">Select all 25 on this page</label>
  <button class="btn btn-sm" type="button" id="all-results">Select all 800 matches</button>
  <details><summary>Recordings on this page</summary><div id="visible" style="max-block-size:4rem; overflow:auto"></div></details>
  <p role="status" id="selection-count">0 of 800 selected</p>
</section>
<script>
const matches = Array.from({length: 800}, (_, i) => i + 1)
const page = matches.slice(0, 25)
const selected = new Set()
const allPage = document.getElementById('page-select')
const rows = document.getElementById('visible')
for (const id of page) {
  const label = document.createElement('label')
  label.className = 'check'
  const input = document.createElement('input')
  input.className = 'checkbox'
  input.type = 'checkbox'
  input.value = id
  label.append(input, 'Recording ' + id)
  rows.append(label)
}
function draw() {
  const count = page.filter(id => selected.has(id)).length
  allPage.checked = count === page.length
  allPage.indeterminate = count > 0 && count < page.length
  for (const input of rows.querySelectorAll('input')) input.checked = selected.has(Number(input.value))
  document.getElementById('selection-count').textContent = selected.size + ' of ' + matches.length + ' selected'
}
function change(ids, checked) {
  for (const id of ids) {
    if (checked) selected.add(id)
    else selected.delete(id)
  }
  draw()
}
allPage.addEventListener('change', () => change(page, allPage.checked))
rows.addEventListener('change', event => change([Number(event.target.value)], event.target.checked))
document.getElementById('all-results').addEventListener('click', () => change(matches, true))
</script>`} />
